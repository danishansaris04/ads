// ================================================================ // PASTE YOUR ADSTERRA VAST TAG URL HERE (from the VAST zone you // create in the Adsterra dashboard): var VAST_TAG_URL = "PASTE_YOUR_ADSTERRA_VAST_TAG_URL_HERE"; // ================================================================ var adContainer = document.getElementById('adContainer'); var contentElement = document.getElementById('contentElement'); var adsManager, adsLoader, adDisplayContainer; function closeOut() { // Nothing further to do here - InterstitialAdActivity's own "Skip" // button (native side) is what actually closes the screen. This // just stops trying to play anything further if the ad fails/ends. try { if (adsManager) adsManager.destroy(); } catch (e) {} } function initAds() { adDisplayContainer = new google.ima.AdDisplayContainer(adContainer, contentElement); adsLoader = new google.ima.AdsLoader(adDisplayContainer); adsLoader.addEventListener( google.ima.AdsManagerLoadedEvent.Type.ADS_MANAGER_LOADED, onAdsManagerLoaded, false); adsLoader.addEventListener( google.ima.AdErrorEvent.Type.AD_ERROR, onAdError, false); var adsRequest = new google.ima.AdsRequest(); adsRequest.adTagUrl = VAST_TAG_URL; adsRequest.linearAdSlotWidth = window.innerWidth; adsRequest.linearAdSlotHeight = window.innerHeight; adsRequest.nonLinearAdSlotWidth = window.innerWidth; adsRequest.nonLinearAdSlotHeight = window.innerHeight; adsLoader.requestAds(adsRequest); } function onAdsManagerLoaded(adsManagerLoadedEvent) { adsManager = adsManagerLoadedEvent.getAdsManager(contentElement); adsManager.addEventListener(google.ima.AdErrorEvent.Type.AD_ERROR, onAdError); adsManager.addEventListener( google.ima.AdEvent.Type.ALL_ADS_COMPLETED, closeOut); try { adDisplayContainer.initialize(); adsManager.init(window.innerWidth, window.innerHeight, google.ima.ViewMode.NORMAL); adsManager.start(); } catch (adError) { closeOut(); } } function onAdError(adErrorEvent) { // No fill / bad tag / network issue - just stop, native "Skip" // button lets the user leave. closeOut(); } // IMA requires a user gesture on some platforms; since this WebView // has setMediaPlaybackRequiresUserGesture(false), we can init on load. window.addEventListener('load', function () { try { initAds(); } catch (e) { closeOut(); } });
